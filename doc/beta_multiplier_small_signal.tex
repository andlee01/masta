\usepackage{tikz}
\usepackage[europeanresistors,americaninductors,americancurrents,siunitx]{circuitikz}

\input{../doc/cmos_small_signal.tex}

\begin{document}
 \begin{circuitikz}

    \draw (0,-2) node[ground](GND){};
    \draw (6,-2) node[ground](GND_M){};
    \draw (0,8) node[vdd](VDD){$V_{DD}sml$};
    \draw (6,8) node[vdd](VDD_M){$V_{DD}sml$};

    % The pmos FET transistor
    \pmosmx{M3}{shift={(0,4)}}{M3}
    \pmos{M4}{shift={(4,4)}}{M4}

    % Vcc Connection
    \draw (SM3) to[short] (VDD);
    \draw (SM4) to[short] (VDD_M);

    % Gate-drain connection
    \draw (GM4) to[short] (GM4 |- DM4) to[short] (DM4);
    \draw (DM4) node[circ] {} node[left] {};
    \draw (GM4) node[circ] {} node[left] {};

    % Connect gates
    \draw (GM3) to[short] (GM4);

    % The nmos FET transistor
    \nmosmx{M1}{shift={(0,0)}}{M1}
    \nmos{M2}{shift={(4,0)}}{M2}

    % Gate-drain connection
    \draw (GM1) to[short] (GM1 |- DM1) to[short] (DM1);
    \draw (DM1) node[circ] {} node[left] {};
    \draw (GM1) node[circ] {} node[left] {};

    % Connect gates
    \draw (GM1) to[short] (GM2);

    % Connect drains
    \draw (DM3) to[short] (DM1);
    \draw (DM4) to[short] (DM2);

    % M1 source to GND
    \draw (SM1) to[short] (GND);

    % Rref
    \draw (SM2) to[R, l^=${R_{ref}}$] ++(0,-2) to[short] (GND_M);
 \end{circuitikz}
 \end{document}
